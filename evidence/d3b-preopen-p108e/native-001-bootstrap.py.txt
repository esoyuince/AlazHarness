from pathlib import Path,PurePosixPath
import os,sys,json,hashlib,tempfile,subprocess
assert os.getresuid()==(23001,)*3 and os.getresgid()==(23003,)*3
assert os.environ.get('WSL_DISTRO_NAME')=='AlazHarness-D3B-P71C'
assert Path('/etc/alaz-p71c-lab-id').read_text().strip()=='AlazHarness-D3B-P71C'
data=json.load(sys.stdin);files=data['files'];assert len(files)<400
assert sum(len(v['text'].encode()) for v in files.values())<12000000
with tempfile.TemporaryDirectory(prefix='alaz-p108e-',dir='/tmp') as tmp:
 root=Path(tmp)
 for name,v in files.items():
  p=PurePosixPath(name);assert not p.is_absolute() and '..' not in p.parts and str(p)==name
  assert p.parts[0] in ('vendor-native','dependencies','packages','scripts') and name.endswith('.py')
  raw=v['text'].encode();assert hashlib.sha256(raw).hexdigest()==v['sha256']
  dest=root/name;dest.parent.mkdir(parents=True,exist_ok=True)
  with dest.open('xb') as f:f.write(raw)
  dest.chmod(0o600)
 (root/'manifest.json').write_text(json.dumps({n:v['sha256'] for n,v in files.items()}))
 env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','HOME':str(root),'WSL_DISTRO_NAME':'AlazHarness-D3B-P71C'}
 r=subprocess.run([sys.executable,'-I','-B',str(root/'scripts/qa/p108e_preopen_probe.py')],cwd=root,env=env,capture_output=True,text=True,timeout=180)
 result={'exitCode':r.returncode,'stdout':r.stdout,'stderr':r.stderr,'sources':{n:v['sha256'] for n,v in files.items()},'sourcesUnchanged':all(hashlib.sha256((root/n).read_bytes()).hexdigest()==v['sha256'] for n,v in files.items()),'uid':os.getuid(),'rootUsed':False,'temporaryRoot':str(root)}
result['temporaryRootRemoved']=not root.exists()
print(json.dumps(result),flush=True)
raise SystemExit(0 if r.returncode==0 and result['sourcesUnchanged'] and result['temporaryRootRemoved'] else 1)
