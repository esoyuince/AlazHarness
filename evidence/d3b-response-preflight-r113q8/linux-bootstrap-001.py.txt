from pathlib import Path,PurePosixPath
import os,json,hashlib,subprocess,sys,tempfile
assert os.getresuid()==(23001,)*3 and os.getresgid()==(23003,)*3
assert os.environ.get('WSL_DISTRO_NAME')=='AlazHarness-D3B-P71C'
assert Path('/etc/alaz-p71c-lab-id').read_text().strip()=='AlazHarness-D3B-P71C'
files=json.load(sys.stdin)['files'];assert len(files)<500
with tempfile.TemporaryDirectory(prefix='alaz-r113q8-code-',dir='/tmp') as tmp:
 root=Path(tmp)
 for name,value in files.items():
  p=PurePosixPath(name);assert not p.is_absolute() and '..' not in p.parts
  data=value['text'].encode();assert hashlib.sha256(data).hexdigest()==value['sha256']
  target=root/name;target.parent.mkdir(parents=True,exist_ok=True)
  with target.open('xb') as f:f.write(data)
 (root/'input.json').write_text(json.dumps({'sources':{n:v['sha256'] for n,v in files.items()}}))
 env={'PATH':'/usr/bin:/bin','HOME':str(root),'LANG':'C.UTF-8','WSL_DISTRO_NAME':'AlazHarness-D3B-P71C'}
 unit=subprocess.run([sys.executable,'-I','-B','-m','unittest','discover','-s',str(root/'tests'),'-p','test_gateway_response_preopen.py','-v'],cwd=root,env=env,capture_output=True,text=True,timeout=40)
 native=subprocess.run([sys.executable,'-I','-B',str(root/'scripts/qa/r113q8_response_preflight_probe.py')],cwd=root,env=env,capture_output=True,text=True,timeout=100)
 report={'unitExitCode':unit.returncode,'unitLog':unit.stdout+unit.stderr,'nativeExitCode':native.returncode,'nativeStdout':native.stdout,'nativeStderr':native.stderr,'sources':{n:v['sha256'] for n,v in files.items()},'sourcesUnchanged':all(hashlib.sha256((root/n).read_bytes()).hexdigest()==v['sha256'] for n,v in files.items()),'uid':os.getuid(),'rootUsed':False,'packageInstalled':False}
report['temporaryRootRemoved']=not root.exists()
print(json.dumps(report),flush=True)
raise SystemExit(0 if unit.returncode==native.returncode==0 and report['sourcesUnchanged'] and report['temporaryRootRemoved'] else 1)
