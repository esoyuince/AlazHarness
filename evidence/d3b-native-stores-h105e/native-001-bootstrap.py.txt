from pathlib import Path,PurePosixPath
import os,json,hashlib,subprocess,sys,tempfile
assert os.getresuid()==(23001,)*3 and os.getresgid()==(23003,)*3
assert os.environ.get('WSL_DISTRO_NAME')=='AlazHarness-D3B-P71C'
assert Path('/etc/alaz-p71c-lab-id').read_text().strip()=='AlazHarness-D3B-P71C'
data=json.load(sys.stdin);files=data['files'];assert 1<len(files)<500
assert sum(len(v['text'].encode()) for v in files.values())<12000000
with tempfile.TemporaryDirectory(prefix='alaz-h105e-code-',dir='/tmp') as tmp:
 root=Path(tmp)
 for name,item in files.items():
  p=PurePosixPath(name);assert not p.is_absolute() and '..' not in p.parts and str(p)==name
  assert p.parts[0] in ('vendor-native','dependencies','scripts') and name.endswith('.py')
  raw=item['text'].encode();assert hashlib.sha256(raw).hexdigest()==item['sha256']
  output=root/name;output.parent.mkdir(parents=True,exist_ok=True)
  with output.open('xb') as f:f.write(raw)
  output.chmod(0o600)
 manifest={'nativePin':data['pin'],'sources':{n:v['sha256'] for n,v in files.items()}}
 with (root/'input-manifest.json').open('x',encoding='utf-8') as f:json.dump(manifest,f);f.write('\n')
 env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','HOME':str(root),'WSL_DISTRO_NAME':'AlazHarness-D3B-P71C'}
 r=subprocess.run([sys.executable,'-I','-B',str(root/'scripts/qa/h105e_store_crash_probe.py')],cwd=root,env=env,capture_output=True,text=True,timeout=460)
 report={'exitCode':r.returncode,'stdout':r.stdout,'stderr':r.stderr,'sourceSha256':manifest['sources'],'sourceUnchanged':all(hashlib.sha256((root/n).read_bytes()).hexdigest()==h for n,h in manifest['sources'].items()),'uid':os.getuid(),'rootUsed':False,'codeRoot':str(root)}
report['temporaryRootRemoved']=not root.exists()
print(json.dumps(report),flush=True)
sys.exit(0 if r.returncode==0 and report['sourceUnchanged'] and report['temporaryRootRemoved'] else 1)
