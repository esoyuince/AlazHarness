from pathlib import Path
import os,sys,tempfile,json,hashlib,importlib.metadata
source=Path(sys.argv[1]);sys.path.insert(0,str(source))
with tempfile.TemporaryDirectory(prefix='h105e-native-discovery-') as tmp:
 home=Path(tmp);os.environ['HERMES_HOME']=str(home)
 def audit(event,args):
  if event.startswith('socket.') and event in ('socket.connect','socket.getaddrinfo','socket.bind'):raise PermissionError('discovery_offline')
 sys.addaudithook(audit)
 from hermes_state import SessionDB
 from gateway.platforms.api_server_run_idempotency import RunIdempotencyStore
 import hermes_state_runtime as runtime
 db=SessionDB(home/'state.db');runs=RunIdempotencyStore(str(home/'runs_idempotency.db'))
 try:
  epoch=runtime.begin_runtime_epoch(db,instance_id='h105e-discovery')
  db.create_session('h105e-session','api')
  row=runtime.admit_session_input(db,epoch=epoch,principal_id='h105e-principal',session_id='h105e-session',request_id='h105e-request',payload={'text':'offline fixture'})
  claim=runtime.claim_session_input(db,epoch=epoch,session_id='h105e-session')
  out,record=runs.reserve('h105e-scope','h105e-key','a'*64,'run_'+('1'*32),{'status':'running'},owner_pid=os.getpid(),owner_started=1)
  assert out=='created' and runs.durable
  e2=runtime.begin_runtime_epoch(db,instance_id='h105e-next')
  recovered=runtime.recover_session_inputs(db,epoch=e2)
  unknown=runtime.get_session_admission(db,admission_id=row['admission_id'])
  assert e2==epoch+1 and recovered==1 and unknown['status']=='unknown'
  try:runtime.claim_session_input(db,epoch=e2,session_id='h105e-session')
  except runtime.RuntimeStoreError as exc:assert str(exc)=='unknown_execution'
  else:raise AssertionError('unknown_replayed')
  print(json.dumps({'phase':'smoke','state':'PASS_NATIVE_STORES','journalState':db._conn.execute('PRAGMA journal_mode').fetchone()[0],'journalRuns':runs._conn.execute('PRAGMA journal_mode').fetchone()[0],'epoch':epoch,'nextEpoch':e2,'recovered':recovered,'status':unknown['status']}))
 finally:runs.close();db.close()
 native={};external={}

  if not f:continue
  p=Path(f).resolve()

  if p.is_relative_to(source):native[p.relative_to(source).as_posix()]=hashlib.sha256(p.read_bytes()).hexdigest()
  elif 'site-packages' in p.parts:
   i=p.parts.index('site-packages');external['/'.join(p.parts[i+1:])]=hashlib.sha256(p.read_bytes()).hexdigest()
 print(json.dumps({'phase':'sources','native':native,'dependencies':external,'sitePackages':str(Path(sys.prefix)/'Lib/site-packages'),'profileRemovedAfterExit':True}))
