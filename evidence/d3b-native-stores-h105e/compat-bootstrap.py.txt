assert os.getresuid()==(23001,)*3 and os.getresgid()==(23003,)*3

results=[];homes=[]
with tempfile.TemporaryDirectory(prefix='alaz-h105e-compat-code-',dir='/tmp') as tmp:
 root=Path(tmp)

   results.append({'mode':mode,'exitCode':r.returncode,'stdout':r.stdout,'stderr':r.stderr})
 report={'cases':results,'sources':{n:v['sha256'] for n,v in files.items()},'sourcesUnchanged':all(hashlib.sha256((root/n).read_bytes()).hexdigest()==v['sha256'] for n,v in files.items())}
