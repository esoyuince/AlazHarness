def open_probe(path, flags=os.O_RDONLY):
    """Open/close only; no data read/write, truncate or create."""
    try:
        fd=os.open(path,flags|os.O_CLOEXEC)
    except OSError as exc:
        return {'opened':False,'errno':exc.errno}
    os.close(fd)
    return {'opened':True,'errno':0}


def kernel_targets():
    return {'synthetic_canary':(str(ROOT.parent/'kernel-denied.txt'),os.O_RDONLY),

def kernel_denials():
    import errno
    values={name:open_probe(*item) for name,item in kernel_targets().items()}
    assert all(not value['opened'] and value['errno']==errno.EACCES for value in values.values()), values
    return values

    before=observer.observation()

    assert all(value['opened'] for value in baseline.values()), baseline

    readonly=[str(ROOT),str(native),'/opt/alaz-f126c1a6/dependencies','/usr/bin',

               for value in readonly)
    writable=[str(home.parent),str(out),str(ROOT.parent/'envhome'),'/dev/null']

            'writable':writable,'protected':[str(ROOT.parent/'kernel-denied.txt')]}

            'denialPairs':{name:{'before':baseline[name],'after':denied[name]} for name in denied},
            'noProductOrOsMethodReplacement':True,'fullProjectionAccepted':False}
