    def tearDown(self):
        for event in self.releases:event.set()
        for pool in (self.runner._executor,self.runner._housekeeping_executor):
            if pool is not None:pool.shutdown(wait=True)
        # Ordinary-cache baseline alone intentionally uses the registry path.
        stats=f.Registry.stats()
        if stats['total_refcounts']:
            f.Registry.close_all_under(self.home)
            for key in tuple(self.goals._DB_CACHE):
                if Path(key).is_relative_to(self.home):del self.goals._DB_CACHE[key]
        OBS.append({'case':self._testMethodName,'registryBeforeFixtureCleanup':stats,'borrowContextReset':self.current() is None})
        Base.tearDown(self)
    def test_no_selected_scope_preserves_default(self):
        self.assertIsNone(self.current())
        import hermes_constants
        home=self.home/'ordinary';home.mkdir();token=hermes_constants.set_hermes_home_override(home)
        try:
            first=self.goals._get_session_db();self.assertIsNotNone(first);self.assertIsNot(first,self.state)
            self.assertIs(self.goals._get_session_db(),first);self.assertEqual(f.Registry.stats()['total_refcounts'],1)
        finally:hermes_constants.reset_hermes_home_override(token)
        self.assert_open()
    def test_three_helper_getters_share_same_original_connection(self):
        with self.watched(),self.scope(lambda:self.state):
            for getter in (self.goals._get_session_db,self.loops._get_session_db,self.heartbeat._get_session_db):self.assertIs(getter(),self.state)
        self.assertEqual(self.goals._DB_CACHE,self.cache_before);self.assertEqual(f.Registry.stats()['total_refcounts'],0);self.assert_open()
    def test_real_goal_loop_heartbeat_persist_into_caller_store(self):
        with self.scope(lambda:self.state):
            self.goals.save_goal('owned',self.goals.GoalState(goal='bounded goal'))
            self.loops.save_loop('owned',self.loops.LoopState(prompt='bounded loop'))

            self.assertEqual(self.goals.load_goal('owned').goal,'bounded goal')
            self.assertEqual(self.loops.load_loop('owned').prompt,'bounded loop')
            self.assertEqual(self.heartbeat.load_heartbeat('owned').prompt,'bounded heartbeat')
        self.assertGreater(self.connections[0].total_changes,0);self.assertEqual(f.Registry.stats()['total_refcounts'],0);self.assertEqual(self.goals._DB_CACHE,self.cache_before)
    def test_expired_copy_refuses_without_registry_fallback(self):
        with self.scope(lambda:self.state):saved=contextvars.copy_context();self.assertIs(saved.run(self.goals._get_session_db),self.state)
        with self.watched(),self.assertRaisesRegex(RuntimeError,'scope_expired'):saved.run(self.goals._get_session_db)
        self.assertEqual(f.Registry.stats()['total_refcounts'],0)
    def test_nested_scope_restores_outer_and_revokes_inner_copy(self):
        with self.scope(lambda:self.state):
            outer=contextvars.copy_context()

                inner=contextvars.copy_context();self.assertIs(self.current(),self.replay)
            self.assertIs(self.current(),self.state);self.assertIs(outer.run(self.current),self.state)
            with self.assertRaisesRegex(RuntimeError,'scope_expired'):inner.run(self.current)
        with self.assertRaisesRegex(RuntimeError,'scope_expired'):outer.run(self.current)
    def test_exception_exits_scope_and_revokes_retained_context(self):
        with self.assertRaisesRegex(ValueError,'explicit'):
            with self.scope(lambda:self.state):saved=contextvars.copy_context();raise ValueError('explicit')

    def test_cancelled_carrier_expires_context_not_still_running_thread(self):
        self.activate();started=threading.Event();release=threading.Event();done=threading.Event();self.releases.append(release);out=[]
        def worker():
            started.set()
            try:
                if not release.wait(8):raise TimeoutError('release missing')
                try:self.goals._get_session_db()
                except RuntimeError as exc:out.append(str(exc))
                else:out.append('unexpected_store_access')
            finally:done.set()
        async def carrier():
            with self.execution_scope(self.authority,nullcontext()):await asyncio.to_thread(worker)

            task=asyncio.create_task(carrier())

            finally:release.set();await asyncio.gather(task,return_exceptions=True)
        f.LOOP.run_until_complete(trial());self.assertTrue(done.wait(5));self.assertEqual(out,['goals_borrowed_scope_expired']);self.assertEqual(f.Registry.stats()['total_refcounts'],0);self.assert_open()
