class TargetOwnerBoundary(unittest.TestCase):
    def setUp(self):
        self.stack=ExitStack();self.addCleanup(self.stack.close);self.events=[]
        self.record=SimpleNamespace(state=object(),response=object(),retained=object());self.mount=object()
        self.receive=self.stack.enter_context(patch.object(launch,'receive',return_value=(self.record,self.mount,'new-owner')))
        import gateway_profile_lease as profiles
        import gateway_store_startup as startup
        self.original_owner_type=startup.NativeStoreStartup
        self.held=Mock();self.owner=Mock();self.owner.expectation=SimpleNamespace(epoch=2)
        self.owner.start.side_effect=lambda:self.mark('start',{'completedChecks':['a','b','c','d','e','f']})
        self.owner.close.side_effect=lambda:self.mark('close');self.owner.status.return_value={'callerLeaseMayBeReleased':True}
        self.held.finish.side_effect=lambda:self.mark('finish')
        self.reserve=self.stack.enter_context(patch.object(profiles,'reserve_profile',side_effect=lambda *a:self.mark('lease',self.held)))
        self.constructor=self.stack.enter_context(patch.object(startup,'NativeStoreStartup',side_effect=lambda *a,**kw:self.mark('construct',self.owner)))
    def mark(self,name,value=None):self.events.append(name);return value
    def test_original_owner_arguments_and_close_before_release(self):

        self.assertEqual(self.constructor.call_args.kwargs,dict(instance_id='new-owner',response_expected=self.record.response,prepared_layout=True,retained_profile=self.record.retained))
        self.assertEqual(result['state'],'PROJECTED_RETAINED_OWNER_CLOSED');self.assertFalse(result['finiteWorkExecuted']);self.assertFalse(result['runtimeAdmissionAuthorized'])
    def test_actual_original_owner_expectation_property_not_read_after_close(self):

        from threading import get_ident
        actor=(os.getpid(),23001,23003,get_ident());fixture=self

            # Actual production descriptor and owned check, not a permissive Mock attribute.
            expectation=fixture.original_owner_type.expectation
            _owned=fixture.original_owner_type._owned
            def __init__(self):
                self._actor=actor;self._held=fixture.held;self._expected=SimpleNamespace(epoch=2);self._stage='NEW'

        with patch.object(original.live.profiles,'_actor',return_value=actor):
            result=launch.target_main()
