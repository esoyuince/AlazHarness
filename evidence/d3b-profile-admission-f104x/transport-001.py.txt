from pathlib import Path,PurePosixPath
import os,json,sys,hashlib,subprocess
D='AlazHarness-D3B-P71C';ROOT=Path('/opt/alaz-f104x');RUN=Path('/srv/alaz-f104x')
assert os.getresuid()==(0,)*3 and os.environ.get('WSL_DISTRO_NAME')==D
assert Path('/etc/alaz-p71c-lab-id').read_text().strip()==D and not Path('/mnt/c/Windows').exists()
assert Path('/proc/self/uid_map').read_text().split()==['0','0','4294967295']
assert not ROOT.exists() and not RUN.exists() and not Path('/sys/fs/cgroup/alaz-f104x').exists()
data=json.load(sys.stdin);files=data['files'];assert 1<len(files)<140
assert sum(len(v['text'].encode()) for v in files.values())<5000000
ROOT.mkdir(mode=0o755)
for name,v in files.items():
 p=PurePosixPath(name);assert str(p)==name and not p.is_absolute() and '..' not in p.parts
 assert p.parts[0] in ('packages','scripts','tests','vendor-native','dependencies') and name.endswith('.py')
 raw=v['text'].encode();assert hashlib.sha256(raw).hexdigest()==v['sha256']
 target=ROOT/name;target.parent.mkdir(parents=True,exist_ok=True)
 with target.open('xb') as stream:stream.write(raw)
 target.chmod(0o644)
manifest={'sources':{n:v['sha256'] for n,v in files.items()},'nativePin':data['nativePin'],'dependencies':'existing ruamel.yaml0.18.16, not installed/updated'}
with (ROOT/'input-manifest.json').open('x',encoding='utf-8') as stream:json.dump(manifest,stream,indent=2);stream.write('\n')
env={'PATH':'/usr/bin:/bin','HOME':'/','LANG':'C.UTF-8','WSL_DISTRO_NAME':D}
u=subprocess.run([sys.executable,'-I','-B','-m','unittest','discover','-s',str(ROOT/'tests'),'-p','test_*.py','-v'],cwd='/',env=env,capture_output=True,text=True,timeout=45)
result={'unitExitCode':u.returncode,'unitStdout':u.stdout,'unitStderr':u.stderr,'sourceHashes':manifest['sources'],'nativePin':data['nativePin']}
if u.returncode==0:
 r=subprocess.run([sys.executable,'-I','-B',str(ROOT/'scripts/qa/f104x_accept.py')],cwd='/',env=env,capture_output=True,text=True,timeout=180)
 result.update(nativeExitCode=r.returncode,nativeStdout=r.stdout,nativeStderr=r.stderr)
result['sourceUnchanged']=all(hashlib.sha256((ROOT/n).read_bytes()).hexdigest()==h for n,h in manifest['sources'].items())
print(json.dumps(result),flush=True)
sys.exit(0 if result.get('nativeExitCode')==0 and result['unitExitCode']==0 and result['sourceUnchanged'] else 1)
