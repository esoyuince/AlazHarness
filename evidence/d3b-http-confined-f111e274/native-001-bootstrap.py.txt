from pathlib import Path,PurePosixPath
import os,json,hashlib,subprocess,sys

assert os.getresuid()==(0,)*3 and os.environ.get('WSL_DISTRO_NAME')==D

assert sum(len(v['text'].encode()) for v in files.values())<12000000
ROOT.mkdir(mode=0o755)

 p=PurePosixPath(n);assert str(p)==n and not p.is_absolute() and '..' not in p.parts
 assert p.parts[0] in ('packages','scripts','tests','vendor-native','dependencies') and n.endswith('.py')

 dest.chmod(0o644)
manifest={'sources':{n:v['sha256'] for n,v in files.items()},'nativePin':data['pin']}

env={'PATH':'/usr/bin:/bin','LANG':'C.UTF-8','HOME':'/','WSL_DISTRO_NAME':D}
r=subprocess.run([sys.executable,'-I','-B','-m','unittest','discover','-s',str(ROOT/'tests'),'-p','test_gateway_store_startup.py','-v'],cwd='/',env=env,capture_output=True,text=True,timeout=45,user=23001,group=23003,extra_groups=[])
result={'unitsExitCode':r.returncode,'unitsStdout':r.stdout,'unitsStderr':r.stderr,'sources':manifest['sources'],'nativePin':data['pin']}
if r.returncode==0:
 r=subprocess.run([sys.executable,'-I','-B',str(ROOT/'scripts/qa/f111e274_accept.py')],cwd='/',env=env,capture_output=True,text=True,timeout=240)
 result.update(nativeExitCode=r.returncode,nativeStdout=r.stdout,nativeStderr=r.stderr)
result['sourcesUnchanged']=all(hashlib.sha256((ROOT/n).read_bytes()).hexdigest()==h for n,h in manifest['sources'].items())

raise SystemExit(0 if result['unitsExitCode']==0 and result.get('nativeExitCode')==0 and result['sourcesUnchanged'] else 1)
