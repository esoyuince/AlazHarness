"""Independent Git-object/source-tar audit. Reads only; NEVER extracts or installs.

The externally reviewed commit, includes and digest are expectations, not
permission to install. This verifier does not import code from the candidate.
"""
from __future__ import annotations
import argparse
from dataclasses import dataclass
import hashlib
import io
import json
import os
from pathlib import Path, PurePosixPath
import re
import stat
import subprocess
import tarfile

MAX_FILE = 2 * 1024 * 1024
MAX_TOTAL = 16 * 1024 * 1024
MAX_ARCHIVE = 32 * 1024 * 1024
MAX_FILES = 512
SHA1 = re.compile(r'[0-9a-f]{40}')
SHA256 = re.compile(r'[0-9a-f]{64}')
PRIVATE = {'runtime', 'artifacts', 'scratch', '.git', '.ssh', '.hermes', '.openclaw', '.alaz-ci'}


class AuditRejected(ValueError):
    pass


def need(ok, code):
    if not ok:
        raise AuditRejected(code)


def safe_name(name):
    need(type(name) is str and 0 < len(name.encode('utf-8')) <= 1024,
         'archive_path_size')
    p = PurePosixPath(name)
    need(name != '.' and not p.is_absolute() and str(p) == name and '..' not in p.parts
         and not any(c in name for c in ('\\', ':', '\x00', '\r', '\n'))
         and all(ord(c) >= 32 for c in name), 'archive_noncanonical_path')
    need(not name.startswith('-') and not any(x.lower() in PRIVATE for x in p.parts),
         'archive_private_path')
    need(not any(x.lower().startswith('.env') for x in p.parts)
         and p.suffix.lower() not in ('.db', '.sqlite', '.sqlite3', '.key', '.pem', '.p12', '.pfx')
         and not re.search(r'(credentials|secrets).*\.json$', p.name, re.I),
         'archive_private_path')
    return name


@dataclass(frozen=True)
class Blob:
    path: str
    mode: str
    oid: str
    data: bytes

    @property
    def sha256(self):
        return hashlib.sha256(self.data).hexdigest()


def git(repo, *args, input=None):
    p = subprocess.run(['git', '--no-replace-objects', '-C', str(repo), *args],
                       input=input, stdout=subprocess.PIPE, stderr=subprocess.PIPE, timeout=40)
    need(p.returncode == 0, 'archive_git_read_failed')
    return p.stdout


def source_blobs(repo, revision, includes):
    need(type(revision) is str and SHA1.fullmatch(revision), 'archive_exact_commit_required')
    need(type(includes) is list and 0 < len(includes) <= 32
         and len(includes) == len(set(includes)), 'archive_include_set')
    for name in includes:
        safe_name(name)
    need(git(repo, 'cat-file', '-t', revision).strip() == b'commit', 'archive_not_commit')
    tree = git(repo, 'ls-tree', '-r', '-z', revision)
    selected = []
    for row in tree.split(b'\0'):
        if not row:
            continue
        metadata, raw_name = row.split(b'\t', 1)
        mode, kind, oid = metadata.decode('ascii').split()
        name = raw_name.decode('utf-8')
        if not any(name == x or name.startswith(x + '/') for x in includes):
            continue
        safe_name(name)
        need(mode in ('100644', '100755') and kind == 'blob', 'archive_source_not_regular')
        selected.append((name, mode, oid))
    need(0 < len(selected) <= MAX_FILES, 'archive_source_count')
    for include in includes:
        need(any(n == include or n.startswith(include + '/') for n, _, _ in selected),
             'archive_include_missing')
    request = ''.join(oid + '\n' for _, _, oid in selected).encode('ascii')
    metadata = git(repo, 'cat-file', '--batch-check', input=request).decode('ascii').splitlines()
    need(len(metadata) == len(selected), 'archive_git_batch_count')
    total = 0
    for row, (_, _, oid) in zip(metadata, selected):
        actual, kind, size = row.split(); size = int(size)
        need(actual == oid and kind == 'blob' and 0 <= size <= MAX_FILE, 'archive_source_bounds')
        total += size
    need(total <= MAX_TOTAL, 'archive_source_bounds')
    stream = io.BytesIO(git(repo, 'cat-file', '--batch', input=request))
    blobs = {}
    for name, mode, oid in selected:
        actual, kind, raw_size = stream.readline().decode('ascii').split()
        size = int(raw_size)
        need(actual == oid and kind == 'blob' and 0 <= size <= MAX_FILE, 'archive_git_object_changed')
        data = stream.read(size)
        need(len(data) == size and stream.read(1) == b'\n', 'archive_git_object_truncated')
        need(hashlib.sha1(b'blob ' + str(size).encode('ascii') + b'\0' + data).hexdigest() == oid,
             'archive_git_blob_digest')
        blobs[name] = Blob(name, mode, oid, data)
    need(not stream.read(1), 'archive_git_extra_output')
    return blobs


def inspect_bytes(data, blobs, prefix, revision):
    """Pure metadata/content audit. No tar extract/extractall/exec operations."""
    need(type(data) is bytes and 1024 <= len(data) <= MAX_ARCHIVE and len(data) % 512 == 0,
         'archive_size_or_block_alignment')
    need(type(revision) is str and SHA1.fullmatch(revision), 'archive_exact_commit_required')
    need(type(prefix) is str and prefix.endswith('/') and not prefix.endswith('//'), 'archive_prefix')
    base = safe_name(prefix[:-1])
    need(type(blobs) is dict and 0 < len(blobs) <= MAX_FILES, 'archive_source_count')
    expected = {}; dirs = {base}; total = 0
    for name, blob in blobs.items():
        safe_name(name)
        need(type(blob) is Blob and blob.path == name and blob.mode in ('100644', '100755')
             and type(blob.data) is bytes and len(blob.data) <= MAX_FILE, 'archive_expected_blob')
        full = prefix + name
        expected[full] = blob
        dirs.update(str(x) for x in PurePosixPath(full).parents if str(x) != '.')
        total += len(blob.data)
    need(total <= MAX_TOTAL, 'archive_source_bounds')
    seen = set(); files = {}; seen_dirs = set(); end = 0
    try:
        with tarfile.open(fileobj=io.BytesIO(data), mode='r:') as archive:
            need(set(archive.pax_headers) <= {'comment'}, 'archive_global_pax')
            need(not archive.pax_headers or archive.pax_headers.get('comment') == revision,
                 'archive_commit_comment')
            for member in archive:
                need(len(seen) < MAX_FILES * 4, 'archive_member_count')
                name = member.name[:-1] if member.isdir() and member.name.endswith('/') else member.name
                safe_name(name)
                need(name not in seen, 'archive_duplicate_member')
                seen.add(name)
                need(member.isreg() or member.isdir(), 'archive_link_or_special_member')
                need(not member.linkname and not member.sparse, 'archive_link_or_sparse')
                need(member.uid == 0 and member.gid == 0
                     and member.uname in ('', 'root') and member.gname in ('', 'root'),
                     'archive_owner_metadata')
                need(set(member.pax_headers) <= {'comment', 'path'}, 'archive_member_pax')
                need('comment' not in member.pax_headers or member.pax_headers['comment'] == revision,
                     'archive_commit_comment')
                need('path' not in member.pax_headers or member.pax_headers['path'].rstrip('/') == name,
                     'archive_pax_path')
                need(0 <= member.size <= MAX_FILE, 'archive_member_bounds')
                if member.isdir():
                    need(name in dirs and member.size == 0 and member.mode == 0o755,
                         'archive_unexpected_directory_or_mode')
                    seen_dirs.add(name)
                else:
                    need(name in expected, 'archive_extra_file')
                    blob = expected[name]
                    need(member.mode == (0o755 if blob.mode == '100755' else 0o644),
                         'archive_file_mode')
                    need(member.size == len(blob.data), 'archive_content_size')
                    handle = archive.extractfile(member)
                    need(handle is not None, 'archive_missing_content')
                    with handle:
                        content = handle.read(MAX_FILE + 1)
                    need(content == blob.data, 'archive_content_mismatch')
                    files[blob.path] = {'sha256': blob.sha256, 'gitBlob': blob.oid,
                                        'size': len(content), 'mode': blob.mode}
                end = max(end, member.offset_data + ((member.size + 511) // 512) * 512)
    except (tarfile.TarError, EOFError, UnicodeError, OSError) as exc:
        raise AuditRejected('archive_malformed') from exc
    need(set(files) == set(blobs), 'archive_missing_file')
    need(seen_dirs == dirs, 'archive_directory_inventory')
    need(len(data) - end >= 1024 and not any(data[end:]), 'archive_nonzero_or_missing_trailer')
    return {'result': 'PASS_ARCHIVE_CONTENT_ONLY', 'revision': revision,
            'archiveSha256': hashlib.sha256(data).hexdigest(), 'archiveBytes': len(data),
            'prefix': prefix, 'filesChecked': len(files), 'directoriesChecked': len(dirs),
            'sourceBytes': total, 'files': files, 'extracted': False, 'installed': False,
            'runtimeDependenciesVerified': False, 'positiveUidTransitionTested': False,
            'fullD3BAccepted': False}


def stable_metadata(before, opened, after, named, *, windows=False):
    # Windows Python may expose creation time through lstat ctime and change
    # time through fstat ctime. Never compare those unlike clocks. Each clock
    # must stay stable; file identity/size/mode/mtime must agree across all four.
    common = lambda s: (s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_mtime_ns)
    stable = (common(before) == common(opened) == common(after) == common(named)
              and before.st_ctime_ns == named.st_ctime_ns
              and opened.st_ctime_ns == after.st_ctime_ns)
    if windows:
        births = [getattr(s, 'st_birthtime_ns', None) for s in (before, opened, after, named)]
        return stable and (all(b is None for b in births) or
                           all(type(b) is int for b in births) and len(set(births)) == 1)
    return stable and before.st_ctime_ns == opened.st_ctime_ns


def stable_archive(path, expected):
    need(type(expected) is str and SHA256.fullmatch(expected), 'archive_external_digest_required')
    p = Path(path)
    before = p.lstat()
    need(stat.S_ISREG(before.st_mode) and not p.is_symlink()
         and 0 < before.st_size <= MAX_ARCHIVE, 'archive_file_bounds')
    with p.open('rb') as handle:
        opened = os.fstat(handle.fileno())
        data = handle.read(MAX_ARCHIVE + 1)
        after = os.fstat(handle.fileno())
    need(stable_metadata(before, opened, after, p.lstat(), windows=os.name == 'nt'),
         'archive_file_changed')
    need(hashlib.sha256(data).hexdigest() == expected, 'archive_digest_mismatch')
    return data


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--repo', required=True, type=Path)
    parser.add_argument('--revision', required=True)
    parser.add_argument('--archive', required=True, type=Path)
    parser.add_argument('--sha256', required=True)
    parser.add_argument('--prefix', required=True)
    parser.add_argument('--include', required=True, action='append')
    parser.add_argument('--output', required=True, type=Path)
    args = parser.parse_args()
    root = Path(__file__).resolve().parents[2]
    dest = args.output
    need(not dest.exists() and not dest.is_symlink()
         and dest.parent.resolve() == (root/'evidence/d3b-install-qa-p83c').resolve(),
         'archive_fresh_own_report_required')
    blobs = source_blobs(args.repo, args.revision, args.include)
    data = stable_archive(args.archive, args.sha256)
    record = inspect_bytes(data, blobs, args.prefix, args.revision)
    record['includes'] = args.include
    record['verifierSha256'] = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    record['scope'] = 'Exact source-only archive integrity; no installation, ownership enforcement or runtime proof'
    with dest.open('x', encoding='utf-8', newline='\n') as handle:
        json.dump(record, handle, sort_keys=True, indent=2); handle.write('\n')
    print(json.dumps({k:v for k,v in record.items() if k != 'files'}))
    return 0


if __name__ == '__main__':
    try:
        raise SystemExit(main())
    except AuditRejected as exc:
        print(json.dumps({'result':'REFUSED','reason':str(exc),'installed':False}), file=__import__('sys').stderr)
        raise SystemExit(2)
