"""Independent real-aiohttp audit of an explicitly pinned readiness module.

Storage/profile/identity boundaries are explicit doubles. Network parsing,
listener bind, keepalive closure and cleanup use real aiohttp sockets. No real
Hermes/model/provider or external address is reached. Accepts trusted LOCAL
source paths only; no downloaded source, arbitrary HTTP target or dynamic code.
"""

ROOT=Path(__file__).resolve().parents[2]
TOKEN='7b'*32  # deterministic synthetic fixture, never a production credential
runtime=None

class StorageDouble:
    def __init__(self):
        self.stage='NEW';self.events=[];self.start_error=None;self.close_error=None
        self.identity_bad=False;self.writer=False;self.verify_error=None
        self._held=object();self._mount=object()
        self.native_state=SimpleNamespace(_conn=object())
        self.native_replay=SimpleNamespace(_conn=object())
        self.want=SimpleNamespace(epoch=1)
        self.service=None
    def status(self):
        return {'state':self.stage,'storageReady':self.stage=='VERIFIED',
                'callerLeaseMayBeReleased':self.stage=='CLOSED'}
    def start(self):
        self.events.append('storage.start')
        if self.start_error:
            self.stage='FAILED';raise self.start_error
        self.stage='VERIFIED';return self.status()
    def _owned(self):
        if self.identity_bad:raise RuntimeError('synthetic_ownership_lost')
    def _drained(self):
        if self.writer:raise RuntimeError('synthetic_active_writer')
    @property
    def state_store(self):self._owned();return self.native_state
    @property
    def replay_store(self):self._owned();return self.native_replay
    @property
    def expectation(self):self._owned();return self.want
    def close(self):
        self.events.append('storage.close')
        if self.service is not None and not self.service._http_gone:
            raise AssertionError('store_closed_before_http_drain')
        if self.close_error:raise self.close_error
        self.stage='CLOSED'

class RealHttpAudit(unittest.IsolatedAsyncioTestCase):
    async def asyncSetUp(self):
        self.owned=[];self.trace=[]
        self.patches=[patch.object(runtime.startup,'NativeStoreStartup',StorageDouble),
            patch.object(runtime.startup.live.profiles,'_actor',side_effect=lambda:(os.getpid(),23001,23003,threading.get_ident())),
            patch.object(runtime.startup.live,'verify_live_stores',side_effect=self.verify)]
        for p in self.patches:p.start()
        self.addCleanup(lambda:[p.stop() for p in reversed(self.patches)])
    def verify(self,held,mount,state,replay,expected):
        self.trace.append('live.check')
        for service,storage in self.owned:
            if storage._held is held and storage.verify_error:raise storage.verify_error
        return {'state':'NATIVE_STORE_CONNECTIONS_VERIFIED','epochObserved':expected.epoch,
                'runtimeAdmissionAuthorized':False}
    def new(self,port=0):
        storage=StorageDouble();service=runtime.StorageReadinessHttp(storage,token=TOKEN,port=port)
        storage.service=service;self.owned.append((service,storage));return service,storage
    async def started(self):
        service,storage=self.new();await service.start();return service,storage
    async def asyncTearDown(self):
        for service,storage in reversed(self.owned):
            # Test-owned abnormal-path cleanup, NOT another product start/retry.
            # Any deliberately injected cleanup exception is restored first.
            if service._runner is not None:
                restore=getattr(service,'_qa_restore_cleanup',None)
                if restore is not None:service._runner.cleanup=restore
                await service._runner.cleanup()
            if getattr(service,'_qa_external_socket',None) is not None:
                service._qa_external_socket.close()
    async def request(self,service,path='/health/ready',*,method='GET',headers=(),body=b'',auth=True,host=None):
        port=service.status()['port']
        reader,writer=await asyncio.wait_for(asyncio.open_connection('127.0.0.1',port),2)
        try:
            rows=[f'{method} {path} HTTP/1.1',f'Host: {host or "127.0.0.1:"+str(port)}']
            if auth:rows.append('Authorization: Bearer '+TOKEN)
            rows += list(headers)
            if body and not any(n.lower().startswith('content-length:') for n in headers):rows.append('Content-Length: '+str(len(body)))
            packet=('\r\n'.join(rows)+'\r\n\r\n').encode()+body
            writer.write(packet);await writer.drain()
            raw=await asyncio.wait_for(reader.readuntil(b'\r\n\r\n'),3)
            lines=raw.decode('iso-8859-1').split('\r\n');status=int(lines[0].split()[1]);parsed={}
            for line in lines[1:]:
                if ':' in line:
                    k,v=line.split(':',1);parsed.setdefault(k.lower(),[]).append(v.strip())
            length=int(parsed.get('content-length',['0'])[0])
            payload=await asyncio.wait_for(reader.readexactly(length),3) if length and method!='HEAD' else b''
            end=await asyncio.wait_for(reader.read(1),3)
            return {'status':status,'headers':parsed,'body':payload,'connectionClosed':end==b''}
        finally:
            writer.close();await writer.wait_closed()
    async def assert_unbound(self,port):
        try:reader,writer=await asyncio.wait_for(asyncio.open_connection('127.0.0.1',port),.4)
        except OSError:return
        else:writer.close();await writer.wait_closed();self.fail('owned listener remained reachable')
    def assert_no_auth_leak(self,reply):
        self.assertNotIn(TOKEN.encode(),reply['body']);self.assertNotIn(TOKEN,str(reply['headers']))
    async def test_authenticated_ready_is_storage_only_and_not_cacheable(self):
        s,st=await self.started();r=await self.request(s);self.assertEqual(r['status'],200)

        self.assertIs(body['storageReady'],True);self.assertIs(body['runtimeAdmissionAuthorized'],False)
        self.assertIs(body['agentRoutesEnabled'],False);self.assertTrue(r['connectionClosed'])
        self.assertEqual(r['headers']['cache-control'],['no-store']);self.assert_no_auth_leak(r)
        self.assertEqual(st.events,['storage.start']);self.assertEqual(len(self.trace),2)
    async def test_liveness_does_not_inspect_or_claim_store_readiness(self):

    async def test_cancellation_during_cleanup_never_closes_stores(self):

        with self.assertRaises(asyncio.CancelledError):await task

def main():
    global runtime
    parser=argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--runtime',type=Path,required=True)
    parser.add_argument('--expected-sha256',required=True)
    parser.add_argument('--output',type=Path,required=True)
    args=parser.parse_args();path=args.runtime.resolve(strict=True)
    raw=path.read_bytes();digest=hashlib.sha256(raw).hexdigest()
    if digest!=args.expected_sha256:raise ValueError('runtime_source_changed_before_test')
    if args.output.exists():raise ValueError('output_must_be_new')
    args.output.mkdir(parents=True)
    sys.path.insert(0,str(ROOT/'packages/claw-hermes'))
    spec=importlib.util.spec_from_file_location('r110x_runtime_snapshot',path)
    runtime=importlib.util.module_from_spec(spec);sys.modules[spec.name]=runtime;spec.loader.exec_module(runtime)
    import aiohttp
    suite=unittest.defaultTestLoader.loadTestsFromTestCase(RealHttpAudit)
    started=time.monotonic()
    with (args.output/'unittest.txt').open('x',encoding='utf-8',newline='\n') as log:
        result=unittest.TextTestRunner(stream=log,verbosity=2).run(suite)
    stable=path.read_bytes()==raw
    report={'result':'PASS_HTTP_ONLY_EXPLICIT_STORAGE_DOUBLES' if result.wasSuccessful() and stable else 'FAIL',
        'total':result.testsRun,'passed':result.testsRun-len(result.errors)-len(result.failures)-len(result.skipped),
        'failures':len(result.failures),'errors':len(result.errors),'skipped':len(result.skipped),
        'failureIds':[t.id() for t,_ in result.failures],'errorIds':[t.id() for t,_ in result.errors],
        'runtimeSha256':digest,'runtimeSourceUnchanged':stable,'qaSha256':hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        'aiohttpVersion':aiohttp.__version__,'elapsedSeconds':round(time.monotonic()-started,3),
        'actualLoopbackHttp':True,'nativeStorageTested':False,'profileAndIdentityDoubled':True,
        'fullConfinementTested':False,'providerCalls':0,'fullD3BAccepted':False}
    with (args.output/'result.json').open('x',encoding='utf-8',newline='\n') as f:json.dump(report,f,indent=2);f.write('\n')
    print(json.dumps(report),flush=True)
    if not result.wasSuccessful():print((args.output/'unittest.txt').read_text(encoding='utf-8')[-14000:],flush=True)
    return 0 if result.wasSuccessful() and stable else 1

if __name__=='__main__':raise SystemExit(main())
