"""Explicit finite cold native-store profile, not a general Hermes home.

Only pinned scaffolding, exact SOUL/cache data, the two databases, their
sidecars and native locks are admitted. Leaf directories are empty; arbitrary
skills/hooks/config/auth content is excluded. This is a metadata/auxiliary-data
snapshot, not DB schema/connection/recovery verification or launch authority.
No database/sidecar/lock descriptor is opened, no file is repaired or created.
Caller owns the native profile lifecycle and checks mount/cgroup exclusivity.
"""
from __future__ import annotations
import hashlib
import json
import os
from pathlib import Path
import stat

PROFILE_LAYOUT='hermes-native-stores-cold-v1'
NATIVE_PIN='eb75627faf532fa6ffbdae59a0d14b0b385fc780'
DATABASES=frozenset(('state.db','runs_idempotency.db'))
LOCKS=frozenset(('gateway.lock','state.db.quarantine.lock','state.db.fts_rebuild.lock'))
DIRECTORIES=frozenset(('cron', 'sessions', 'logs', 'logs/curator', 'memories', 'pairing', 'hooks', 'image_cache', 'audio_cache', 'skills', 'cache'))
AUX_FILES=frozenset(('SOUL.md','cache/schema_columns.json'))
CORE_FILES=DATABASES|LOCKS|AUX_FILES
SIDECARS=frozenset(n+s for n in DATABASES for s in ('-journal','-wal','-shm'))
ALLOWED=CORE_FILES|DIRECTORIES|SIDECARS
REQUIRED=CORE_FILES|DIRECTORIES
MAX_FILE_BYTES=2*1024**3
SOUL_SHA256='36c1f5a2e92cd1d018311eaf4c8f1e8886672eae78212e033c681d0e3d5d506f'
SOUL_BYTES=667
CACHE_SHA256='8f4094e34ba4c09f2109fd04376c5f4a75597f31a8f9b7a75e527ce1ab87e114'
MAX_CACHE_BYTES=65536

class NativeLayoutRejected(ValueError):
    pass

def require(ok,reason):
    if not ok:raise NativeLayoutRejected(reason)

def metadata(info):
    return (info.st_dev,info.st_ino,info.st_mode,info.st_uid,info.st_gid,
            info.st_nlink,info.st_size,info.st_mtime_ns,info.st_ctime_ns)

def _private(info,uid,gid,device,directory):
    kind=stat.S_ISDIR if directory else stat.S_ISREG
    require(kind(info.st_mode) and info.st_dev==device and info.st_uid==uid
            and info.st_gid==gid and stat.S_IMODE(info.st_mode)==(0o700 if directory else 0o600)
            and (directory or info.st_nlink==1),'native_layout_private_object')
    require(directory or 0<=info.st_size<=MAX_FILE_BYTES,'native_layout_file_limit')
    return metadata(info)

def _children(path,allowed):
    # Reject during iteration, before any unbounded materialization/sort.
    found={}
    with os.scandir(path) as stream:
        for entry in stream:
            require(entry.name in allowed and entry.name not in found,'native_layout_unexpected_entry')
            found[entry.name]=entry.stat(follow_symlinks=False)
    return found

def _snapshot(home,uid,gid,device):
    observed={'profile':_private(home.lstat(),uid,gid,device,True)}
    pending=[('',home)]
    while pending:
        relative,parent=pending.pop()
        allowed={Path(n).name for n in ALLOWED if str(Path(n).parent).replace('\\','/')==(relative or '.')}
        entries=_children(parent,allowed)
        required={Path(n).name for n in REQUIRED if str(Path(n).parent).replace('\\','/')==(relative or '.')}
        require(required<=set(entries),'native_layout_required_entry_missing')
        for child,info in sorted(entries.items()):
            name=relative+'/'+child if relative else child
            directory=name in DIRECTORIES
            observed[name]=_private(info,uid,gid,device,directory)
            require(not getattr(parent/child,'is_junction',lambda:False)(),'native_layout_link')
            if directory:pending.append((name,parent/child))
            elif name in DATABASES:require(info.st_size>=512,'native_layout_unprepared_database')
            elif name in LOCKS:require(info.st_size<=16384,'native_layout_lock_size')
            elif name=='SOUL.md':require(info.st_size==SOUL_BYTES,'native_layout_soul_size')
            elif name=='cache/schema_columns.json':require(0<info.st_size<=MAX_CACHE_BYTES,'native_layout_cache_size')
    return observed

def _read_aux(home,name,observed,maximum):
    expected=observed[name]
    # These two files must never alias live SQLite state or ownership locks.
    require(expected[:2] not in {observed[n][:2] for n in DATABASES|LOCKS|SIDECARS if n in observed},
            'native_layout_aux_alias')
    fd=os.open(home/name,os.O_RDONLY|os.O_CLOEXEC|os.O_NOFOLLOW)
    try:
        require(metadata(os.fstat(fd))==expected,'native_layout_aux_changed')
        raw=os.read(fd,maximum+1)
        require(len(raw)==expected[6] and len(raw)<=maximum,'native_layout_aux_changed')
        require(metadata(os.fstat(fd))==expected,'native_layout_aux_changed')
        return raw
    finally:os.close(fd)

def _cache(raw):
    def pairs(items):
        result={}
        for key,value in items:
            require(key not in result,'native_layout_cache_duplicate_key');result[key]=value
        return result
    value=json.loads(raw,object_pairs_hook=pairs,
                     parse_constant=lambda _:(_ for _ in ()).throw(NativeLayoutRejected('native_layout_cache_nonfinite')))
    canonical=json.dumps(value,sort_keys=True,separators=(',',':'),ensure_ascii=True).encode()
    require(hashlib.sha256(canonical).hexdigest()==CACHE_SHA256,'native_layout_cache_content')

def inspect_profile(profile,*,uid,gid,device):
    """Finite store-only admission snapshot; mount and native lease are separate.

    Rechecks reject observed mutation, not every ABA race. Caller must not run
    this against an untrusted concurrently mutable same-UID profile. All SQL
    connection checks must use existing connections; this function opens only
    the exact non-DB SOUL/cache paths, never SQLite files/sidecars/lock files.
    """
    try:
        require(all(type(n) is int and n>0 for n in (uid,gid,device)),'native_layout_identity')
        require(isinstance(profile,(str,Path)),'native_layout_home')
        home=Path(profile)
        require(home.is_absolute() and home==home.resolve(strict=True)
                and not any(p.is_symlink() or getattr(p,'is_junction',lambda:False)() for p in (home,*home.parents)),
                'native_layout_canonical_profile')
        first=_snapshot(home,uid,gid,device)
        soul=_read_aux(home,'SOUL.md',first,SOUL_BYTES)
        require(hashlib.sha256(soul).hexdigest()==SOUL_SHA256,'native_layout_soul_content')
        _cache(_read_aux(home,'cache/schema_columns.json',first,MAX_CACHE_BYTES))
        require(first==_snapshot(home,uid,gid,device),'native_layout_changed')
        return first
    except NativeLayoutRejected:raise
    except (OSError,ValueError,TypeError,KeyError,AttributeError,OverflowError,RecursionError):
        raise NativeLayoutRejected('native_layout_unavailable') from None
