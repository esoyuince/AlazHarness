from pathlib import Path
import importlib.metadata as m,json,sys
from packaging.requirements import Requirement
from packaging.markers import default_environment
from packaging.utils import canonicalize_name
r=json.loads(Path(sys.argv[1]).read_text(encoding='utf-8'));roots=set(r['distributions'])-{'pywin32'}
env=default_environment();env.update(os_name='posix',sys_platform='linux',platform_system='Linux',platform_machine='x86_64',python_version='3.12',python_full_version='3.12.3',implementation_name='cpython',implementation_version='3.12.3',extra='')
queue=sorted(roots);done={};missing={};omitted=[]
while queue:
 name=queue.pop(0);key=canonicalize_name(name)
 if key in done:continue
 try:d=m.distribution(name)
 except m.PackageNotFoundError:missing[name]=name;continue
 record={'name':d.metadata['Name'],'version':d.version,'requiresPython':d.metadata.get('Requires-Python'),'dependencies':[]};done[key]=record
 for raw in (d.requires or []):
  dep=Requirement(raw)
  if dep.marker and not dep.marker.evaluate(env):omitted.append({'from':name,'dependency':raw});continue
  record['dependencies'].append(raw)
  try:
   child=m.distribution(dep.name);assert child.version in dep.specifier,(name,dep.name,child.version,str(dep.specifier))
  except m.PackageNotFoundError:missing[dep.name]=str(dep.name)+str(dep.specifier);continue
  queue.append(dep.name)
try:pip=m.version('pip')
except m.PackageNotFoundError:pip=None
result={'target':env,'observedRootDistributions':sorted(roots),'explicitExcludedPlatformPackage':'pywin32','lockedInstalledClosure':list(done.values()),'missingTargetOnly':missing,'excludedByMarker':omitted,'pipVersionInImmutableRuntime':pip}
Path(sys.argv[2]).write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'locked':{v['name']:v['version'] for v in done.values()},'missingTargetOnly':missing,'pip':pip}))
