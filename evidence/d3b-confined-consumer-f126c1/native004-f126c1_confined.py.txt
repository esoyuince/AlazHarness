def connection_errno(kind,address):
    with socket.socket(kind,socket.SOCK_STREAM) as s:
        s.settimeout(1)
        try:s.connect(address)
        except OSError as ex:return ex.errno
    return 0


def fields():
    return {k:v.split() for row in Path('/proc/self/status').read_text().splitlines()
            for k,sep,v in [row.partition(':')] if sep}
