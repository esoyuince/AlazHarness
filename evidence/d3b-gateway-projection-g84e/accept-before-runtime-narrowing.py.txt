def require(ok,message):
    if not ok:raise RuntimeError(message)

def digest(path):return hashlib.sha256(path.read_bytes()).hexdigest()

def fileproof(path):
    s=path.stat();return {'sha256':digest(path),'dev':s.st_dev,'ino':s.st_ino,
                         'mode':stat.S_IMODE(s.st_mode),'uid':s.st_uid,'gid':s.st_gid}

def dump(path,data):
    with path.open('x',encoding='utf-8') as f:json.dump(data,f,indent=2);f.write('\n')
    path.chmod(0o600)

def observed(pid):
    p=Path('/proc')/str(pid);fields={}
    for line in (p/'status').read_text().splitlines():
        key,_,value=line.partition(':')
        if key in ('Uid','Gid','Groups','NoNewPrivs','Seccomp',*launch.CAPS):fields[key]=value.split()
    return {'pid':pid,'fields':fields,'namespaces':{n:os.readlink(p/'ns'/n) for n in ('user','mnt','net','ipc')},
            'uidMap':(p/'uid_map').read_text().split(),'gidMap':(p/'gid_map').read_text().split(),
            'cmdline':(p/'cmdline').read_bytes().decode().split('\0')[:-1]}

def readline_bounded(process,seconds=30):
    end=time.monotonic()+seconds;data=bytearray();fd=process.stdout.fileno()
    while len(data)<32768:
        remaining=end-time.monotonic()
        require(remaining>0,'gateway_ready_timeout')
        require(select.select([fd],[],[],remaining)[0],'gateway_ready_timeout')
        b=os.read(fd,1)
        require(bool(b),'gateway_exited_before_ready')
        data.extend(b)
        if b==b'\n':return json.loads(data)
    raise RuntimeError('gateway_ready_output_bound')

print(json.dumps({'uid':os.getuid(),'gid':os.getgid(),'connections':out,'canaryReadable':True}))

    r=subprocess.run([PY,'-I','-B','-c',code],input=json.dumps({'paths':paths,'port':tcp_port,'abstract':abstract,'canary':str(canary)}).encode(),
          user=23001,group=23003,extra_groups=[],capture_output=True,cwd='/',env=ENV,timeout=5)
    require(r.returncode==0 and not r.stderr,'host_positive_controls_failed')
    return json.loads(r.stdout)
