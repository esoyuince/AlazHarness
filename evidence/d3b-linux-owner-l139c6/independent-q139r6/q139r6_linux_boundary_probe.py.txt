def validate(raw,digest):

    z=zipfile.ZipFile(io.BytesIO(raw));names=z.namelist();infos=z.infolist()
    if len(names)!=len(set(names)) or len(names)>1200 or sum(v.file_size for v in infos)>60_000_000:raise ValueError('transport_shape')
    m=json.loads(z.read('transport-manifest.json'))
    if m['nativeFeature']!=FEATURE or set(names)!=set(m['files'])|{'transport-manifest.json'}:raise ValueError('transport_set')
    for name,h in m['files'].items():

        if p.is_absolute() or '..' in p.parts or chr(92) in name or ':' in name or stat.S_ISLNK(z.getinfo(name).external_attr>>16):raise ValueError('transport_path')
        allowed=name.startswith(('native/','harness/packages/claw-hermes/','harness/config/')) or name in ('accepted-dependencies.json','harness/scripts/run_store_preopen.py','harness/scripts/run_response_preopen.py')
        if not allowed or hashlib.sha256(z.read(name)).hexdigest()!=h:raise ValueError('transport_member')
    return z,m
