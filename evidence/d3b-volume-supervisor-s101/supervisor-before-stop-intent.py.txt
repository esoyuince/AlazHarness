"""Trusted Linux state-volume supervisor, explicit clean close or retained fence.

Not a Gateway launcher: the fixed worker mounts and holds an existing ext4
image only. No formatting, model callback, root shell, repair or force-detach.
All callers and the installation must be trusted host root. A root-only fence
is durable BEFORE attachment; after controller death it blocks blind reuse.
Cgroup membership is root-controlled; every volume user must join that scope.
"""
from __future__ import annotations
import ctypes
from dataclasses import asdict
import errno
import hashlib
import json
import os
from pathlib import Path
import re
import secrets
import select
import stat
import struct
import subprocess
import sys
import threading
import time
from state_image_lease import ImagePin, acquire_image_lease

ROOT = Path(__file__).resolve().parents[2]
INFO = struct.Struct('<QQQQQIIII64s64s32sQQ')
GET_STATUS = 0x4C05
CONFIGURE = 0x4C0A
GET_FREE = 0x4C82
AUTOCLEAR = 4
CGROUP2_MAGIC = 0x63677270
_RETAINED = {}  # strong references: exceptions/GC MUST NOT release a live lease


class SupervisorRejected(RuntimeError):
    pass


def require(condition, reason):
    if not condition:
        raise SupervisorRejected(reason)


def actor():
    require(sys.platform == 'linux', 'supervisor_linux_required')
    require(os.uname().machine == 'x86_64' and sys.byteorder == 'little', 'supervisor_abi')
    require(os.getresuid() == (0, 0, 0) and os.getresgid() == (0, 0, 0), 'supervisor_host_root')
    require(Path('/proc/self/uid_map').read_text().split() == ['0', '0', '4294967295']
            and Path('/proc/self/gid_map').read_text().split() == ['0', '0', '4294967295'],
            'supervisor_initial_ids')
    require(threading.current_thread() is threading.main_thread(), 'supervisor_main_thread')
    return os.getpid(), threading.get_ident()


def trusted(path, *, directory=False):
    path = Path(path)
    require(path.is_absolute() and path == path.resolve(strict=True), 'supervisor_canonical_path')
    for p in (path, *path.parents):
        s = p.lstat()
        require(s.st_uid == 0 and not stat.S_IMODE(s.st_mode) & 0o022
                and not stat.S_ISLNK(s.st_mode), 'supervisor_untrusted_path')
    s = path.stat()
    require((stat.S_ISDIR(s.st_mode) if directory else stat.S_ISREG(s.st_mode)),
            'supervisor_object_type')
    return s.st_dev, s.st_ino


def fs_type(path):
    lib = ctypes.CDLL(None, use_errno=True)
    lib.statfs.argtypes = [ctypes.c_char_p, ctypes.c_void_p]
    lib.statfs.restype = ctypes.c_int
    data = ctypes.create_string_buffer(256)
    require(lib.statfs(os.fsencode(path), data) == 0, 'supervisor_filesystem_unreadable')
    return ctypes.c_long.from_buffer(data).value


def populated(text):
    require(type(text) is str and 0 < len(text) < 8192, 'supervisor_cgroup_events')
    values = {}
    for line in text.splitlines():
        parts = line.split()
        require(len(parts) == 2 and parts[0] not in values
                and re.fullmatch(r'[0-9]+', parts[1]) is not None, 'supervisor_cgroup_events')
        values[parts[0]] = int(parts[1])
    require(values.get('populated') in (0, 1), 'supervisor_cgroup_events')
    return values['populated'] == 1


def loop_status(fd):
    import fcntl
    data = bytearray(INFO.size)
    try:
        fcntl.ioctl(fd, GET_STATUS, data, True)
    except OSError as exc:
        if exc.errno == errno.ENXIO:
            return None
        raise SupervisorRejected('supervisor_loop_status_unknown') from None
    v = INFO.unpack(data)
    # kernel huge_encode_dev/new_encode_dev, not a pathname comparison
    major, minor = (v[0] >> 8) & 0xfff, (v[0] & 0xff) | ((v[0] >> 12) & 0xfff00)
    return {'device': os.makedev(major, minor), 'inode': v[1], 'offset': v[3],
            'sizeLimit': v[4], 'number': v[5], 'flags': v[8], 'encryption': v[6]}


def open_loop(number):
    require(type(number) is int and 0 <= number < 65536, 'supervisor_loop_number')
    fd = os.open('/dev/loop' + str(number), os.O_RDWR | os.O_CLOEXEC | os.O_NOFOLLOW)
    s = os.fstat(fd)
    if not (stat.S_ISBLK(s.st_mode) and s.st_rdev == os.makedev(7, number)):
        os.close(fd)
        raise SupervisorRejected('supervisor_loop_device')
    return fd


def attachments(pin):
    """Global loop identity inventory, not a path-based or namespace-only check."""
    def names():
        found = sorted(p.name for p in Path('/sys/class/block').iterdir()
                       if re.fullmatch(r'loop[0-9]+', p.name))
        require(0 < len(found) <= 4096, 'supervisor_loop_inventory_bound')
        return found
    before = names(); matches = []
    for name in before:
        fd = open_loop(int(name[4:]))
        try:
            row = loop_status(fd)
            if row is not None and (row['device'], row['inode']) == (pin.device, pin.inode):
                matches.append(row)
        finally:
            os.close(fd)
    require(before == names(), 'supervisor_loop_inventory_changed')
    return matches


def read_line(process, seconds=15):
    deadline = time.monotonic() + seconds; data = bytearray(); fd = process.stdout.fileno()
    while len(data) < 16384:
        left = deadline - time.monotonic()
        require(left > 0 and select.select([fd], [], [], left)[0], 'supervisor_worker_timeout')
        item = os.read(fd, 1)
        require(item, 'supervisor_worker_eof')
        data.extend(item)
        if item == b'\n':
            value = json.loads(data)
            require(type(value) is dict, 'supervisor_worker_message')
            return value
    raise SupervisorRejected('supervisor_worker_output_bound')


class VolumeSupervisor:
    """One shot, process/thread-bound. finish() is guarded, never a finally unlock.

    Failure after durable fence creation retains lease and fence. Caller must
    keep the supervisor alive, inspect the effect, drain ONLY owned users and
    explicitly finish. No destructor, automatic retry or force cleanup exists.
    A crash leaves the fence for separately authorized manual reconciliation.
    """
    def __init__(self):
        raise SupervisorRejected('supervisor_use_begin')

    @classmethod
    def begin(cls, pin, cgroup_parent, mountpoint):
        who = actor()
        require(type(pin) is ImagePin and pin.uid == pin.gid == 0, 'supervisor_private_root_image')
        trusted(Path(pin.path).parent, directory=True)
        cg = Path(cgroup_parent); trusted(cg, directory=True)
        require(cg.is_relative_to('/sys/fs/cgroup') and cg != Path('/sys/fs/cgroup')
                and fs_type(cg) == CGROUP2_MAGIC, 'supervisor_dedicated_cgroup2_parent')
        target = Path(mountpoint); target_id = trusted(target, directory=True)
        require(not os.path.ismount(target) and not any(target.iterdir()), 'supervisor_empty_mountpoint')
        lease = acquire_image_lease(pin)
        instance = object.__new__(cls)
        instance.pin, instance.lease, instance.who = pin, lease, who
        instance.target, instance.target_id = target, target_id
        instance.token = secrets.token_hex(16)
        instance.scope = cg / ('volume-' + instance.token)
        instance.scope_id = None
        instance.fence = Path(pin.path + '.volume-owner.json')
        instance.fence_id = None
        instance.loop_fd = None; instance.loop_number = None; instance.worker = None
        instance.stage = 'PREPARING'; instance.spawn_uncertain = False
        body = (json.dumps({'version': 1, 'token': instance.token, 'controllerPid': who[0],
                            'image': asdict(pin), 'scope': str(instance.scope),
                            'mountpoint': str(target)}, sort_keys=True) + '\n').encode()
        instance.fence_bytes = body
        created = False
        try:
            require(not attachments(pin), 'supervisor_image_already_attached')
            fd = os.open(instance.fence, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW | os.O_CLOEXEC, 0o600)
            created = True
            _RETAINED[instance.token] = instance
            with os.fdopen(fd, 'wb') as stream:
                stream.write(body); stream.flush(); os.fsync(stream.fileno())
            instance.fence_id = trusted(instance.fence)
            instance._sync_parent()
            instance.scope.mkdir(mode=0o755)
            instance.scope_id = trusted(instance.scope, directory=True)
            require(instance._empty_scope(), 'supervisor_scope_not_empty')
            instance.stage = 'PREPARED'
            return instance
        except BaseException:
            if not created:
                lease.close()
            else:
                instance.stage = 'QUARANTINED'
            raise

    def _sync_parent(self):
        fd = os.open(self.fence.parent, os.O_RDONLY | os.O_DIRECTORY | os.O_CLOEXEC | os.O_NOFOLLOW)
        try: os.fsync(fd)
        finally: os.close(fd)

    def _owned(self):
        require(actor() == self.who, 'supervisor_wrong_actor')
        require(self.stage != 'CLOSED', 'supervisor_closed')
        self.lease.confirm()
        require(trusted(self.fence) == self.fence_id
                and self.fence.stat().st_nlink == 1
                and self.fence.stat().st_size == len(self.fence_bytes)
                and stat.S_IMODE(self.fence.stat().st_mode) == 0o600
                and self.fence.read_bytes() == self.fence_bytes, 'supervisor_fence_changed')
        require(self.scope_id is not None and trusted(self.scope, directory=True) == self.scope_id
                and fs_type(self.scope) == CGROUP2_MAGIC, 'supervisor_scope_changed')

    def _empty_scope(self):
        require(trusted(self.scope, directory=True) == self.scope_id, 'supervisor_scope_changed')
        text = (self.scope / 'cgroup.events').read_text()
        return not populated(text) and not (self.scope / 'cgroup.procs').read_text().strip()

    def attach(self):
        self._owned()
        require(self.stage == 'PREPARED', 'supervisor_attach_once')
        require(self._empty_scope() and not attachments(self.pin), 'supervisor_previous_users')
        self.stage = 'ATTACHING'
        import fcntl
        control = os.open('/dev/loop-control', os.O_RDONLY | os.O_CLOEXEC | os.O_NOFOLLOW)
        try:
            s = os.fstat(control)
            require(stat.S_ISCHR(s.st_mode) and s.st_rdev == os.makedev(10, 237), 'supervisor_loop_control')
            number = fcntl.ioctl(control, GET_FREE)
        finally:
            os.close(control)
        # Never hand the flock-bearing open description to the loop driver.
        # The kernel's backing-file reference must not keep the flock alive.
        self.loop_fd = open_loop(number); self.loop_number = number
        require(loop_status(self.loop_fd) is None, 'supervisor_selected_loop_busy')
        image = os.open(self.pin.path, os.O_RDWR | os.O_CLOEXEC | os.O_NOFOLLOW)
        try:
            s = os.fstat(image)
            require((s.st_dev, s.st_ino, s.st_size) == (self.pin.device, self.pin.inode, self.pin.size),
                    'supervisor_backing_changed')
            self.lease.confirm()
            info = INFO.pack(0, 0, 0, 0, 0, 0, 0, 0, AUTOCLEAR, b'', b'', b'', 0, 0)
            config = struct.pack('<II', image, 0) + info + bytes(64)
            require(INFO.size == 232 and len(config) == 304, 'supervisor_loop_abi')
            fcntl.ioctl(self.loop_fd, CONFIGURE, config)
        finally:
            os.close(image)
        expected = {'device': self.pin.device, 'inode': self.pin.inode, 'offset': 0,
                    'sizeLimit': 0, 'number': number, 'flags': AUTOCLEAR, 'encryption': 0}
        require(loop_status(self.loop_fd) == expected and attachments(self.pin) == [expected],
                'supervisor_attached_identity')
        self.stage = 'ATTACHED'
        return '/dev/loop' + str(number)

    def start(self):
        self._owned()
        require(self.stage == 'ATTACHED' and self.worker is None, 'supervisor_start_once')
        require(trusted(self.target, directory=True) == self.target_id
                and not any(self.target.iterdir()), 'supervisor_mountpoint_changed')
        for name in ('packages/claw-hermes/state_volume_supervisor.py',
                     'packages/claw-hermes/state_image_lease.py', 'scripts/run_state_volume_worker.py'):
            trusted(ROOT / name)
        python = Path(sys.executable).resolve(); trusted(python)
        self.stage = 'STARTING'; self.spawn_uncertain = True
        self.worker = subprocess.Popen([str(python), '-I', '-B', str(ROOT/'scripts/run_state_volume_worker.py')],
            cwd='/', env={'PATH': '/usr/bin:/bin', 'LANG': 'C.UTF-8'}, stdin=subprocess.PIPE,
            stdout=subprocess.PIPE, stderr=subprocess.PIPE, bufsize=0, close_fds=True)
        self.spawn_uncertain = False
        message = {'version': 1, 'token': self.token, 'controllerPid': self.who[0],
                   'scope': str(self.scope), 'scopeIdentity': list(self.scope_id),
                   'mountpoint': str(self.target), 'targetIdentity': list(self.target_id),
                   'loopNumber': self.loop_number, 'imageDevice': self.pin.device,
                   'imageInode': self.pin.inode}
        self.worker.stdin.write((json.dumps(message)+'\n').encode()); self.worker.stdin.flush()
        ready = read_line(self.worker)
        require(ready.get('state') == 'MOUNTED' and ready.get('pid') == self.worker.pid
                and ready.get('token') == self.token and ready.get('loopNumber') == self.loop_number,
                'supervisor_ready_mismatch')
        require(os.readlink('/proc/'+str(self.worker.pid)+'/ns/mnt') == ready['mountNamespace']
                and ready['mountNamespace'] != os.readlink('/proc/self/ns/mnt'), 'supervisor_worker_not_private')
        require(str(self.worker.pid) in (self.scope/'cgroup.procs').read_text().split(),
                'supervisor_worker_outside_scope')
        require(not os.path.ismount(self.target), 'supervisor_mount_leaked_to_host')
        self.stage = 'RUNNING'
        return ready

    def stop(self):
        self._owned()
        require(self.worker is not None and self.stage in ('RUNNING', 'STOP_REFUSED'), 'supervisor_stop_state')
        require(self.worker.poll() is None, 'supervisor_worker_already_exited')
        self.worker.stdin.write(('STOP '+self.token+'\n').encode()); self.worker.stdin.flush()
        self.stage = 'STOPPING'
        reply = read_line(self.worker)
        require(reply.get('token') == self.token and reply.get('pid') == self.worker.pid,
                'supervisor_stop_reply')
        if reply.get('state') == 'BUSY':
            self.stage = 'STOP_REFUSED'
            raise SupervisorRejected('supervisor_users_still_active')
        require(reply.get('state') == 'UNMOUNTED', 'supervisor_unmount_unconfirmed')
        self.worker.wait(timeout=10)
        require(self.worker.returncode == 0, 'supervisor_worker_unclean_exit')
        self.stage = 'STOPPED'
        return reply

    def finish(self):
        self._owned()
        require(not self.spawn_uncertain, 'supervisor_spawn_unknown')
        require(self.worker is None or self.worker.poll() is not None, 'supervisor_worker_alive')
        if self.worker is not None:
            self.worker.wait(timeout=0)
        require(self._empty_scope(), 'supervisor_users_still_active')
        require(trusted(self.target, directory=True) == self.target_id
                and not os.path.ismount(self.target) and not any(self.target.iterdir()),
                'supervisor_host_mount_remains')
        if self.loop_fd is not None:
            fd, self.loop_fd = self.loop_fd, None
            os.close(fd)  # AUTOCLEAR; may remain busy in another namespace
        require(not attachments(self.pin), 'supervisor_attachment_remains')
        self._owned()
        require(self._empty_scope(), 'supervisor_scope_repopulated')
        self.scope.rmdir()  # kernel also refuses a live/nonempty cgroup
        # Do not reset/recreate a changed fence. All users and loop are gone.
        require(trusted(self.fence) == self.fence_id and self.fence.read_bytes() == self.fence_bytes,
                'supervisor_fence_changed')
        self.fence.unlink(); self._sync_parent()
        self.lease.close(); self.stage = 'CLOSED'; _RETAINED.pop(self.token, None)
        if self.worker is not None:
            for stream in (self.worker.stdin, self.worker.stdout, self.worker.stderr):
                if stream is not None: stream.close()
        return {'state': 'CLOSED', 'ownedUsersGone': True, 'loopAttachmentGone': True,
                'fenceRemovedAfterCleanup': True, 'leaseReleasedLast': True,
                'realHermesGatewayTested': False, 'fullD3BAccepted': False}
